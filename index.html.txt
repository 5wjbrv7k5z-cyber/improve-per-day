        timerInterval = window.setInterval(() => {
          remainingSeconds -= 1;
          if (remainingSeconds <= 0) {
            remainingSeconds = 0;
            stopTimer();
            Sound.success();
            notify('Focus sprint complete! Lock in your daily progress.');
          } else {
            updateTimerDisplay();
          }
        }, 1000);
        updateTimerDisplay();
      }

      function renderSevenDayMatrix(slot) {
        const container = document.getElementById('sevenDayMatrix');
        if (!container) return;
        container.innerHTML = '';

        const today = getDateKey();
        const doneSet = new Set(slot.completedDates);
        const dayLetters = ['S', 'M', 'T', 'W', 'T', 'F', 'S'];

        for (let i = 6; i >= 0; i -= 1) {
          const key = addDays(today, -i);
          const [y, m, d] = key.split('-').map(Number);
          const dt = new Date(y, m - 1, d);
          const isDone = doneSet.has(key);
          const isToday = i === 0;

          const cell = document.createElement('div');
          cell.className = 'flex flex-col items-center gap-1';

          const dotStyle = isDone
            ? 'bg-emerald-400 border-emerald-300 shadow-[0_0_8px_rgba(52,211,153,0.75)]'
            : isToday
              ? 'bg-obsidian-800 border-emerald-400/70'
              : 'bg-obsidian-950 border-emerald-500/20';

          cell.innerHTML = `
            <span class="w-2.5 h-2.5 rounded-full border ${dotStyle}"></span>
            <span class="font-mono text-[9px] ${isToday ? 'text-emerald-300 font-bold' : 'text-emerald-100/45'}">${dayLetters[dt.getDay()]}</span>
          `;
          container.appendChild(cell);
        }
      }

      function setActiveHabit({ title, mode, categoryLabel }) {
        const slot = state.slots[state.activeSlot];

      function handleCompleteToday() {
        const slot = state.slots[state.activeSlot];
        if (!slot) return;
        const today = getDateKey();
        if (slot.completedDates.includes(today)) return;

        slot.completedDates.push(today);
        slot.completedDates = Array.from(new Set(slot.completedDates)).sort();

        const metrics = computeMetrics(slot);
        slot.bestStreak = Math.max(slot.bestStreak || 0, metrics.bestStreak);

        saveState(state);
        render();
        Sound.success();
        spawnBurst(document.getElementById('completeTodayBtn'));

        const milestone = getMilestoneEncouragement(metrics.currentStreak);
        if (milestone) {
          const badgeEl = document.getElementById('milestoneModalBadge');
          const headingEl = document.getElementById('milestoneModalHeading');
          const msgEl = document.getElementById('milestoneModalMessage');
          if (badgeEl) badgeEl.textContent = milestone.badge;
          if (headingEl) headingEl.textContent = milestone.heading;
          if (msgEl) msgEl.textContent = milestone.message;
          openModal('milestoneModal');
        } else {
          notify(`Logged "${slot.title}" · Streak: ${metrics.currentStreak}d`);
        }
      }

      function handleUndoToday() {
        const slot = state.slots[state.activeSlot];
        if (!slot) return;
        const today = getDateKey();
        const idx = slot.completedDates.indexOf(today);
        if (idx === -1) return;
        slot.completedDates.splice(idx, 1);
        saveState(state);
        render();
        Sound.tap();
        notify('Today\'s check-in undone');
      }

      /**
       * Resets current streak (from today backwards) on slip-up and displays the 20 push-ups message in English
       */
      function handleConfirmSlipUp() {
        const slot = state.slots[state.activeSlot];
        if (!slot) return;

        const today = getDateKey();
        let cursor = today;
        const removeSet = new Set();
        while (slot.completedDates.includes(cursor)) {
          removeSet.add(cursor);
          cursor = addDays(cursor, -1);
        }
        slot.completedDates = slot.completedDates.filter((d) => !removeSet.has(d));

        state.slipUps.unshift({
          id: `${Date.now()}`,
          dateKey: today,
          habitTitle: slot.title,
        });
        state.slipUps = state.slipUps.slice(0, 20);

        saveState(state);
        closeModal('slipConfirmModal');
        render();
        Sound.resetAlert();

        const penaltyTextEl = document.getElementById('pushupsPenaltyText');
        if (penaltyTextEl) {
          penaltyTextEl.textContent = PUSHUPS_PENALTY_MESSAGE;
        }
        openModal('pushupsPenaltyModal');
      }

      function handleProceedRealCheckout() {
        Sound.tap();
        const url = (state.checkoutUrl || '').trim();
        if (!url) {
          notify('Add your real checkout URL first in PRO_CHECKOUT_URL above.');
          document.getElementById('checkoutUrlInput')?.focus();
          return;
        }
        try {
          const parsed = new URL(url);
          if (parsed.protocol === 'https:' || parsed.protocol === 'http:') {
            window.open(parsed.toString(), '_blank', 'noopener,noreferrer');
          } else {
            notify('Checkout URL must start with https:// or http://');
          }
        } catch (e) {
          notify('Please enter a valid checkout URL (https://...)');
        }
      }

      function handleSaveCheckoutUrl() {
        const input = document.getElementById('checkoutUrlInput');
        const raw = (input?.value || '').trim();
        if (!raw) {
          state.checkoutUrl = '';
          saveState(state);
          updateCheckoutModalUI();
          notify('Checkout URL cleared');
          return;
        }
        try {
          const u = new URL(raw);
          if (u.protocol !== 'https:' && u.protocol !== 'http:') {
            notify('Use a valid http:// or https:// URL');
            return;
          }
          state.checkoutUrl = u.toString();
          saveState(state);
          updateCheckoutModalUI();
          Sound.tap();
          notify('Real checkout URL saved');
        } catch (e) {
          notify('Invalid URL format');
        }
      }

      /**
       * JSON Export / Import / Wipe
       */
      function handleExportJson() {
        Sound.tap();
        try {
          const blob = new Blob([JSON.stringify(state, null, 2)], { type: 'application/json' });
          const url = URL.createObjectURL(blob);
          const a = document.createElement('a');
          a.href = url;
          a.download = `improve-per-day-backup-${getDateKey()}.json`;
          document.body.appendChild(a);
          a.click();
          document.body.removeChild(a);
          URL.revokeObjectURL(url);
          notify('JSON backup downloaded');
        } catch (e) {
          notify('Could not export backup in this environment');
        }
      }

      function handleImportJsonFile(file) {
        if (!file) return;
        const reader = new FileReader();
        reader.onload = () => {
          try {
            const parsed = JSON.parse(String(reader.result || ''));
            if (!parsed || !Array.isArray(parsed.slots) || parsed.slots.length !== 3) {
              notify('Invalid backup file format');
              return;
            }
            window.localStorage.setItem(STORAGE_KEY, JSON.stringify(parsed));
            state = loadState();
            remainingSeconds = state.timerMinutes * 60;
            render();
            Sound.success();
            notify('Backup restored successfully');
          } catch (err) {
            notify('Could not parse JSON backup');
          }
        };
        reader.readAsText(file);
      }

      function handleWipeAll() {
        stopTimer();
        state = defaultState();
        remainingSeconds = state.timerMinutes * 60;
        memoryState = null;
        try {
          window.localStorage.removeItem(STORAGE_KEY);
        } catch (e) {}
        render();
        Sound.tap();
        notify('All local data cleared');
      }

          if (f) handleImportJsonFile(f);
          e.target.value = '';
        });
        document.getElementById('wipeAllDataBtn')?.addEventListener('click', handleWipeAll);

        // Backdrop & Escape close
        ['proModal', 'milestoneModal', 'slipConfirmModal', 'pushupsPenaltyModal'].forEach((id) => {
          const modal = document.getElementById(id);
          modal?.addEventListener('click', (e) => {
            if (e.target === modal) closeModal(id);
          });
        });

        document.addEventListener('keydown', (e) => {
          if (e.key === 'Escape') {
            ['proModal', 'milestoneModal', 'slipConfirmModal', 'pushupsPenaltyModal'].forEach(closeModal);
          }
        });

        document.addEventListener('visibilitychange', () => {
          if (!document.hidden) render();
        });
      }

      window.addEventListener('DOMContentLoaded', () => {
        bindEvents();
        render();
      });
    })();
  </script>
